1//! cypera — reusable documentation/reference generation for the 9aia
2//! org. The shared successor to the per-repo `docs/gen-*.py`
3//! scripts: one tool renders the crate map, CLI reference, and
4//! config reference pages every sibling docs site needs, driven by
5//! a per-repo `cypera.toml` discovered upward from `--repo`.
6//!
7//! ```text
8//! cypera gen crate-map [--repo <path>] [--config <path>] [--out <file>] [--layer <crate=layer>]…
9//! cypera gen cli       [--repo <path>] [--config <path>] [--input <file|->] [--out <file>]
10//! cypera gen config    [--repo <path>] [--config <path>] [--file <rs>]… [--out <file>]
11//! cypera gen api       [--repo <path>] [--config <path>] [--out <dir>] [--target-dir <dir>]
12//! cypera check         [--repo <path>] [--config <path>] [--json]
13//! ```
14//!
15//! All generation logic lives in `cypera-core` (pure); this crate
16//! owns argv, `cypera.toml` discovery, filesystem walks, process
17//! spawning, and stdout.

67/// The `gen` generators.
68#[derive(Debug, Subcommand)]
69pub enum Gen {
70    /// Crate map — workspace `Cargo.toml`s → crate/layer/purpose
71    /// table (+ per-crate module sections when entry files exist).
72    /// `cypera.toml [crate_map]` supplies layer overrides, the
73    /// layer order, the default layer, splice targets, and the
74    /// crate-api page.
75    CrateMap {
76        /// Repo root (dir containing the workspace `Cargo.toml`).
77        /// Default: current directory.
78        #[arg(long, default_value = ".")]
79        repo: PathBuf,
80        /// Explicit `cypera.toml` path (default: discovered upward
81        /// from `--repo`).
82        #[arg(long)]
83        config: Option<PathBuf>,
84        /// Write a full page to this file instead of stdout. When
85        /// the file exists and contains a `<!-- crate-map -->`
86        /// marker, only the table after the marker is replaced.
87        /// Overrides `[crate_map]` splices/crate_api.
88        #[arg(long)]
89        out: Option<PathBuf>,
90        /// Layer override `crate-name=layer` (repeatable) —
91        /// overrides `[crate_map.layers]` and the heuristic.
92        #[arg(long = "layer")]
93        layers: Vec<String>,
94    },
95    /// CLI reference — two modes:
96    ///
97    /// - `--input <file|->`: wrap `gen-cli-docs` markdown
98    ///   (clap-markdown) in a reference page.
99    /// - otherwise: the `docs/gen-cli-docs.py` port — walk
100    ///   `<bin> <prefix>… <sub>… --help` recursively and emit the
101    ///   same page. The command spec comes from `--package`/`--bin`/
102    ///   `--prefix` flags or `[cli]` in `cypera.toml`.
103    Cli {
104        /// Repo root — cargo build + `cypera.toml` discovery base.
105        #[arg(long, default_value = ".")]
106        repo: PathBuf,
107        /// Explicit `cypera.toml` path.
108        #[arg(long)]
109        config: Option<PathBuf>,
110        /// `gen-cli-docs` markdown file, or `-` for stdin. When
111        /// given, wrap mode (no help walk).
112        #[arg(long)]
113        input: Option<String>,
114        /// Cargo package to `cargo build -q -p` first (config
115        /// `[cli].package` fallback).
116        #[arg(long)]
117        package: Option<String>,
118        /// Binary under `target/debug/` — or a PATH executable
119        /// when `--package` is unset (`[cli].bin`).
120        #[arg(long)]
121        bin: Option<String>,
122        /// Argv prefix before the walked subcommand path, e.g.
123        /// `run` for the driver `run` passthrough (repeatable;
124        /// `[cli].prefix` fallback).
125        #[arg(long = "prefix")]
126        prefixes: Vec<String>,
127        /// Page title override (`[cli].title` fallback; default is
128        /// the python scripts' `Command-Line Help for …` header).
129        #[arg(long)]
130        title: Option<String>,
131        /// Subcommand path to skip in the `--help` walk, space-
132        /// joined (`driver exec`; repeatable; `[cli].skip` fallback).
133        #[arg(long = "skip")]
134        skips: Vec<String>,
135        /// Write to this file instead of stdout
136        /// (`[cli].out` fallback, then stdout).
137        #[arg(long)]
138        out: Option<PathBuf>,
139    },
140    /// Config reference — two modes:
141    ///
142    /// - `--file <rs>…`: scan `pub struct` doc comments and render
143    ///   the generic page (`--only`/`--title` apply).
144    /// - otherwise: the `docs/gen-config-docs.py` port — the
145    ///   curated `[config]` spec in `cypera.toml` (struct set,
146    ///   section titles, type vocabulary, prose sections).
147    Config {
148        /// Repo root — `cypera.toml` discovery + relative `file`
149        /// resolution base.
150        #[arg(long, default_value = ".")]
151        repo: PathBuf,
152        /// Explicit `cypera.toml` path.
153        #[arg(long)]
154        config: Option<PathBuf>,
155        /// Rust source file to scan (repeatable) — legacy mode.
156        #[arg(long = "file")]
157        files: Vec<PathBuf>,
158        /// Only document these struct names (repeatable). Legacy
159        /// mode; in `[config]` mode filters struct sections.
160        #[arg(long = "only")]
161        only: Vec<String>,
162        /// Page title. Legacy mode default: `Config reference`;
163        /// `[config]` mode: overrides `[config].title`.
164        #[arg(long)]
165        title: Option<String>,
166        /// Write to this file instead of stdout
167        /// (`[config].out` fallback, then stdout).
168        #[arg(long)]
169        out: Option<PathBuf>,
170    },
171    /// Rustdoc API site — the `docs/gen-api*.py` port:
172    /// `cargo doc --workspace --no-deps` in the repo, the real
173    /// target dir resolved via `cargo metadata` (respects a
174    /// configured `target-dir`), `<target>/doc` copied into the
175    /// docs site's `public/api/`, and a styled crate index
176    /// written to `index.html`. The output is a gitignored
177    /// build artifact — `check` has nothing to freshness-gate.
178    /// `[api]` in cypera.toml supplies `out`/`prefix`/`title`/
179    /// `target_dir`; every field is optional.
180    Api {
181        /// Repo root — the `cargo doc`/`cargo metadata` cwd.
182        /// A `<NAME>_ROOT` env var (NAME = the cypera.toml dir's
183        /// basename, uppercased — the python `TAMIAS_ROOT`/
184        /// `MYCO_ROOT`/… convention) overrides it, which is how
185        /// detached-sha doc pipelines point at the real checkout.
186        /// Default: current directory.
187        #[arg(long, default_value = ".")]
188        repo: PathBuf,
189        /// Explicit `cypera.toml` path (default: discovered upward
190        /// from `--repo`).
191        #[arg(long)]
192        config: Option<PathBuf>,
193        /// API output *dir* override — replaces
194        /// `[api].out`/`docs/public/api`; a relative path resolves
195        /// against the cwd (run from the docs worktree to land in
196        /// a detached docs checkout, the python `DOCS` anchor).
197        #[arg(long)]
198        out: Option<PathBuf>,
199        /// Cargo target dir for the doc build — exported as
200        /// `CARGO_TARGET_DIR` when unset (`[api].target_dir`
201        /// fallback; the myco `target/docs-api` isolation).
202        #[arg(long)]
203        target_dir: Option<PathBuf>,
204    },
205}

303/// Canonicalize `repo` and locate `cypera.toml`: the explicit
304/// `--config` path when given, else upward discovery from the repo
305/// root (the file may sit at the repo root or an ancestor — e.g. a
306/// workspace-level config). Returns `(config_path, parsed)`.
307fn discover_config(repo: &Path, explicit: Option<&Path>) -> Result<Option<(PathBuf, RepoConfig)>> {
308    let path = match explicit {
309        Some(p) => p.to_path_buf(),
310        None => {
311            let root = repo
312                .canonicalize()
313                .with_context(|| format!("--repo {} does not exist", repo.display()))?;
314            let mut dir = Some(root.as_path());
315            let mut found = None;
316            while let Some(d) = dir {
317                let cand = d.join("cypera.toml");
318                if cand.is_file() {
319                    found = Some(cand);
320                    break;
321                }
322                dir = d.parent();
323            }
324            match found {
325                Some(p) => p,
326                None => return Ok(None),
327            }
328        }
329    };
330    let text = std::fs::read_to_string(&path)
331        .with_context(|| format!("cannot read {}", path.display()))?;
332    let cfg = RepoConfig::parse(&path.display().to_string(), &text)?;
333    Ok(Some((path, cfg)))
334}

336/// Directory relative paths in `cypera.toml` resolve against —
337/// the config file's parent (repo root in the common case).
338fn config_dir(config_path: &Option<(PathBuf, RepoConfig)>) -> PathBuf {
339    config_path
340        .as_ref()
341        .and_then(|(p, _)| p.parent().map(Path::to_path_buf))
342        .unwrap_or_else(|| PathBuf::from("."))
343}

345/// Resolve a config-relative path: try repo-root-relative first
346/// (the documented convention), then config-dir-relative.
347fn resolve_path(repo: &Path, base: &Path, rel: &str) -> PathBuf {
348    let p = Path::new(rel);
349    if p.is_absolute() {
350        return p.to_path_buf();
351    }
352    let under_repo = repo.join(p);
353    if under_repo.exists() {
354        return under_repo;
355    }
356    base.join(p)
357}

359/// Repo display name — the `origin` remote's repo basename when it
360/// resolves (detached/gate worktrees live under hashed dirs like
361/// `worktrees-9aia-myco-<sha>` where the dir name is not the repo
362/// name), else the directory basename.
363pub(crate) fn repo_name(root: &Path) -> String {
364    if let Ok(out) = std::process::Command::new("git")
365        .arg("-C")
366        .arg(root)
367        .args(["remote", "get-url", "origin"])
368        .output()
369    {
370        if out.status.success() {
371            let url = String::from_utf8_lossy(&out.stdout);
372            if let Some(seg) = url.trim().rsplit('/').next() {
373                let seg = seg.strip_suffix(".git").unwrap_or(seg);
374                // scp-like `host:repo` (no slash) segments carry a
375                // ':' — not a repo name.
376                if !seg.is_empty() && !seg.contains(':') {
377                    return seg.to_string();
378                }
379            }
380        }
381    }
382    root.file_name()
383        .map(|n| n.to_string_lossy().into_owned())
384        .unwrap_or_else(|| "repo".to_string())
385}

389/// Parse `crate=layer` overrides into `(name, layer)` pairs.
390fn parse_layers(raw: &[String]) -> Result<Vec<(String, String)>> {
391    raw.iter()
392        .map(|s| {
393            s.split_once('=')
394                .map(|(n, l)| (n.to_string(), l.to_string()))
395                .with_context(|| format!("--layer expects crate=layer, got `{s}`"))
396        })
397        .collect()
398}

400/// Expand `[workspace] members` entries — plain paths pass
401/// through; `dir/*` globs expand to every subdir containing a
402/// `Cargo.toml`.
403fn expand_members(root: &Path, members: &[String]) -> Result<Vec<PathBuf>> {
404    let mut out = Vec::new();
405    for m in members {
406        if let Some((dir, _)) = m.rsplit_once("/*") {
407            let base = root.join(dir);
408            let mut subs: Vec<PathBuf> = std::fs::read_dir(&base)
409                .with_context(|| format!("cannot read member glob dir {}", base.display()))?
410                .filter_map(|e| e.ok().map(|e| e.path()))
411                .filter(|p| p.is_dir() && p.join("Cargo.toml").is_file())
412                .collect();
413            subs.sort();
414            if subs.is_empty() {
415                bail!(
416                    "member glob `{m}` matched no crates under {}",
417                    base.display()
418                );
419            }
420            out.extend(subs);
421        } else {
422            out.push(root.join(m));
423        }
424    }
425    Ok(out)
426}

428/// Scan a crate dir's entry file (`src/lib.rs`, else `src/main.rs`)
429/// for `//!` docs and `mod` declarations.
430fn scan_entry(crate_dir: &Path) -> Option<EntryScan> {
431    for (file, is_lib) in [("src/lib.rs", true), ("src/main.rs", false)] {
432        let p = crate_dir.join(file);
433        if let Ok(src) = std::fs::read_to_string(&p) {
434            return Some(scan_entry_source(&src, is_lib));
435        }
436    }
437    None
438}

440/// Build the [`cypera_core::CrateMap`] + rendered page for a repo —
441/// shared by `gen crate-map` and `check` freshness. `extra_layers`
442/// are `--layer` CLI overrides (they win over `[crate_map.layers]`).
443fn build_crate_map(
444    repo: &Path,
445    cfg: Option<&RepoConfig>,
446    extra_layers: &[String],
447) -> Result<(cypera_core::CrateMap, String)> {
448    let root = repo
449        .canonicalize()
450        .with_context(|| format!("--repo {} does not exist", repo.display()))?;
451    let ws_path = root.join("Cargo.toml");
452    let ws_text = std::fs::read_to_string(&ws_path)
453        .with_context(|| format!("cannot read {}", ws_path.display()))?;
454    let ws = WorkspaceManifest::parse(&ws_path.display().to_string(), &ws_text)?;
455
456    // Bare `[package]` root = single-crate repo.
457    let member_dirs = if ws.members().is_empty() {
458        vec![root.clone()]
459    } else {
460        expand_members(&root, ws.members())?
461    };
462
463    let repo_name = repo_name(&root);
464
465    let mut members = Vec::new();
466    let mut entries = Vec::new();
467    for dir in &member_dirs {
468        let manifest_path = dir.join("Cargo.toml");
469        let text = std::fs::read_to_string(&manifest_path)
470            .with_context(|| format!("cannot read {}", manifest_path.display()))?;
471        let rel = dir
472            .strip_prefix(&root)
473            .unwrap_or(dir)
474            .to_string_lossy()
475            .into_owned();
476        let rel = if rel.is_empty() { ".".to_string() } else { rel };
477        let mut manifest = CrateManifest::parse(&manifest_path.display().to_string(), &text)?;
478        let entry = scan_entry(dir);
479        // Purpose resolution — the `gen-crate-map.py` `description()`
480        // order: literal `[package] description` → first `//!`
481        // paragraph of the entry file (tidied into one cell) →
482        // `[workspace.package] description` when the crate opts in
483        // (`description.workspace = true`) → `"—"`.
484        let mut purpose = manifest.description().to_string();
485        if purpose.is_empty() {
486            if let Some(e) = &entry {
487                purpose = purpose_from_entry_doc(&e.doc);
488            }
489            if purpose.is_empty() {
490                purpose = if manifest.description_inherited() {
491                    ws.workspace_description().unwrap_or("").to_string()
492                } else {
493                    "—".to_string()
494                };
495            }
496        }
497        manifest.package.description = Some(toml::Value::String(purpose));
498        members.push((rel.clone(), manifest));
499        if let Some(scan) = entry {
500            entries.push((rel, scan));
501        }
502    }
503
504    // `--layer` flags first — first-match wins, so CLI beats config.
505    let cli_overrides = parse_layers(extra_layers)?;
506    let mut overrides = cli_overrides;
507    if let Some(c) = cfg {
508        overrides.extend(c.layer_overrides());
509    }
510    let order = cfg.map_or(&[][..], |c| c.crate_map.order.as_slice());
511    let default_layer = cfg.and_then(|c| c.crate_map.default_layer.as_deref());
512    let map = crate_map_from_workspace(
513        &repo_name,
514        &ws,
515        &members,
516        &entries,
517        &overrides,
518        order,
519        default_layer,
520    )?;
521    let page = render_crate_map(&map);
522    Ok((map, page))
523}

525/// Apply `[crate_map]` splices + crate_api to the repo — the
526/// `gen-crate-map.py` wrapper behavior (`--out`-less mode).
527fn apply_crate_map_outputs(
528    root: &Path,
529    base: &Path,
530    cfg: &RepoConfig,
531    map: &cypera_core::CrateMap,
532) -> Result<usize> {
533    let table = crate_map_table(map);
534    let mut written = 0;
535    for splice in &cfg.crate_map.splices {
536        let target = resolve_path(root, base, &splice.target);
537        let existing = std::fs::read_to_string(&target)
538            .with_context(|| format!("cannot read splice target {}", target.display()))?;
539        let spliced = if splice.first_table {
540            splice_first_table(&existing, &table)
541        } else {
542            let marker = splice.marker.as_deref().unwrap_or("<!-- crate-map -->");
543            splice_marked(&existing, marker, &table).with_context(|| {
544                format!("splice marker `{marker}` not found in {}", target.display())
545            })?
546        };
547        std::fs::write(&target, spliced)
548            .with_context(|| format!("cannot write {}", target.display()))?;
549        println!("crate map spliced into {}", target.display());
550        written += 1;
551    }
552    if let Some(api) = &cfg.crate_map.crate_api {
553        let target = resolve_path(root, base, &api.out);
554        std::fs::write(&target, crate_api_page(&api.site, map))
555            .with_context(|| format!("cannot write {}", target.display()))?;
556        println!("crate api written to {}", target.display());
557        written += 1;
558    }
559    Ok(written)
560}

562fn gen_crate_map(
563    repo: &Path,
564    config: Option<&Path>,
565    out: Option<&Path>,
566    layers: &[String],
567) -> Result<i32> {
568    let discovered = discover_config(repo, config)?;
569    let cfg = discovered.as_ref().map(|(_, c)| c);
570    let (map, page) = build_crate_map(repo, cfg, layers)?;
571
572    match out {
573        Some(path) => {
574            // `<!-- crate-map -->` marker → splice only the table
575            // (the gen-crate-map.py in-place convention); otherwise
576            // the existing generated page is replaced wholesale.
577            if let Ok(existing) = std::fs::read_to_string(path) {
578                if let Some(i) = existing
579                    .lines()
580                    .position(|l| l.contains("<!-- crate-map -->"))
581                {
582                    let table = crate_map_table(&map);
583                    let mut lines: Vec<String> = existing.lines().map(str::to_string).collect();
584                    let mut k = i + 1;
585                    while k < lines.len() && lines[k].starts_with('|') {
586                        k += 1;
587                    }
588                    lines.splice(i + 1..k, table.lines().map(str::to_string));
589                    std::fs::write(path, lines.join("\n") + "\n")?;
590                    println!("crate map spliced into {}", path.display());
591                    return Ok(0);
592                }
593            }
594            std::fs::write(path, &page)
595                .with_context(|| format!("cannot write {}", path.display()))?;
596            println!("crate map written to {}", path.display());
597        }
598        None => {
599            if let Some((_, rc)) = &discovered {
600                let root = repo.canonicalize()?;
601                let base = config_dir(&discovered);
602                let n = apply_crate_map_outputs(&root, &base, rc, &map)?;
603                // The `out` page is a declared output like the
604                // splices — `check` freshness verifies it, so it
605                // must be written even when splices ran.
606                if let Some(out_rel) = &rc.crate_map.out {
607                    let target = resolve_path(&root, &base, out_rel);
608                    std::fs::write(&target, &page)
609                        .with_context(|| format!("cannot write {}", target.display()))?;
610                    println!("crate map written to {}", target.display());
611                } else if n == 0 {
612                    print!("{page}");
613                }
614            } else {
615                print!("{page}");
616            }
617        }
618    }
619    Ok(0)
620}

624/// CLI-verb flags shared between clap argv and the `[cli]` config
625/// fallback — everything `gen cli` can take from either source.
626struct CliOpts<'a> {
627    input: Option<&'a str>,
628    package: Option<&'a str>,
629    bin: Option<&'a str>,
630    prefixes: &'a [String],
631    title: Option<&'a str>,
632    skips: &'a [String],
633    out: Option<&'a Path>,
634}
635
636fn gen_cli(repo: &Path, config: Option<&Path>, opts: CliOpts<'_>) -> Result<i32> {
637    let CliOpts {
638        input,
639        package,
640        bin,
641        prefixes,
642        title,
643        skips,
644        out,
645    } = opts;
646    // Wrap mode — `gen-cli-docs` markdown in a page.
647    if let Some(input) = input {
648        let body = if input == "-" {
649            std::io::read_to_string(std::io::stdin())?
650        } else {
651            std::fs::read_to_string(input).with_context(|| format!("cannot read {input}"))?
652        };
653        let page = render_cli_page(title.unwrap_or("Commands"), &body);
654        return emit(page, out).map(|_| 0);
655    }
656
657    let discovered = discover_config(repo, config)?;
658    let file_cfg = discovered.as_ref().and_then(|(_, c)| c.cli.clone());
659    // Flag > `[cli]` config > error.
660    let spec = resolve_cli_spec(file_cfg.as_ref(), package, bin, prefixes, skips)?;
661    let title = title
662        .map(str::to_string)
663        .or_else(|| file_cfg.as_ref().and_then(|c| c.title.clone()));
664    let root = repo
665        .canonicalize()
666        .with_context(|| format!("--repo {} does not exist", repo.display()))?;
667    // Config-declared `out` resolves repo-root-relative (the
668    // splices/crate_api convention); an explicit `--out` flag stays
669    // cwd-relative.
670    let emit_to = match out {
671        Some(p) => Some(p.to_path_buf()),
672        None => file_cfg.as_ref().and_then(|c| {
673            c.out
674                .as_ref()
675                .map(|rel| resolve_path(&root, &config_dir(&discovered), rel))
676        }),
677    };
678
679    let mut spec = spec;
680    spec.title = title;
681    let page = cli_help_page(&root, &spec)?;
682    emit_to_path(page, emit_to.as_deref())?;
683    Ok(0)
684}

686/// Merge `[cli]` config with flag overrides into a [`CliConfig`].
687fn resolve_cli_spec(
688    file_cfg: Option<&CliConfig>,
689    package: Option<&str>,
690    bin: Option<&str>,
691    prefixes: &[String],
692    skips: &[String],
693) -> Result<CliConfig> {
694    let mut spec = file_cfg.cloned().unwrap_or(CliConfig {
695        package: None,
696        bin: String::new(),
697        prefix: Vec::new(),
698        title: None,
699        out: None,
700        skip: Vec::new(),
701    });
702    if let Some(p) = package {
703        spec.package = Some(p.to_string());
704    }
705    if let Some(b) = bin {
706        spec.bin = b.to_string();
707    }
708    if !prefixes.is_empty() {
709        spec.prefix = prefixes.to_vec();
710    }
711    if !skips.is_empty() {
712        spec.skip = skips.to_vec();
713    }
714    if spec.bin.is_empty() {
715        bail!("gen cli: no `bin` — pass --bin or set `[cli].bin` in cypera.toml");
716    }
717    Ok(spec)
718}

720/// `cargo build -q -p <pkg>` once, then walk `<bin> <prefix>…
721/// <path>… --help` recursively — the `gen-cli-docs.py` IO half.
722fn cli_help_page(root: &Path, spec: &CliConfig) -> Result<String> {
723    let bin_path = if let Some(pkg) = &spec.package {
724        let status = std::process::Command::new("cargo")
725            .args(["build", "-q", "-p", pkg])
726            .current_dir(root)
727            .status()
728            .with_context(|| format!("cargo build -p {pkg} failed to spawn"))?;
729        if !status.success() {
730            bail!("cargo build -p {pkg} failed");
731        }
732        // Honor CARGO_TARGET_DIR (deploy pipelines isolate builds —
733        // e.g. docs/scripts/deploy-docs.sh's target/gate); a
734        // relative value resolves against the repo root like cargo.
735        let target = match std::env::var_os("CARGO_TARGET_DIR") {
736            Some(dir) => {
737                let p = PathBuf::from(dir);
738                if p.is_absolute() {
739                    p
740                } else {
741                    root.join(p)
742                }
743            }
744            None => root.join("target"),
745        };
746        target.join("debug").join(&spec.bin)
747    } else {
748        PathBuf::from(&spec.bin)
749    };
750
751    fn help_text(root: &Path, bin: &Path, spec: &CliConfig, path: &[String]) -> Result<String> {
752        let mut cmd = std::process::Command::new(bin);
753        cmd.args(&spec.prefix)
754            .args(path)
755            .arg("--help")
756            .current_dir(root);
757        let out = cmd
758            .output()
759            .with_context(|| format!("cannot spawn {}", bin.display()))?;
760        if !out.status.success() {
761            let label = if path.is_empty() {
762                "<root>".to_string()
763            } else {
764                path.join(" ")
765            };
766            bail!(
767                "help failed for {label}:\n{}",
768                String::from_utf8_lossy(&out.stderr)
769            );
770        }
771        Ok(String::from_utf8_lossy(&out.stdout).into_owned())
772    }
773
774    let mut nodes: Vec<HelpNode> = Vec::new();
775    let mut seen: HashSet<Vec<String>> = HashSet::new();
776
777    fn walk(
778        root: &Path,
779        bin: &Path,
780        spec: &CliConfig,
781        path: Vec<String>,
782        nodes: &mut Vec<HelpNode>,
783        seen: &mut HashSet<Vec<String>>,
784    ) -> Result<()> {
785        if !seen.insert(path.clone()) {
786            return Ok(());
787        }
788        // `[cli].skip` — passthrough surfaces (`disable_help_flag`)
789        // have no `--help` to walk; the parent still lists them.
790        if spec
791            .skip
792            .iter()
793            .any(|s| s.split_whitespace().eq(path.iter().map(String::as_str)))
794        {
795            return Ok(());
796        }
797        let text = help_text(root, bin, spec, &path)?;
798        let (usage, about, sections) = parse_help(&text);
799        // Subcommand names for the recursion — python reads them
800        // from the parsed Commands section before pushing.
801        let subs: Vec<String> = sections
802            .iter()
803            .find(|(n, _)| n == "Commands")
804            .map(|(_, entries)| {
805                entries
806                    .iter()
807                    .filter_map(|(sig, _)| sig.split_whitespace().next().map(str::to_string))
808                    .filter(|s| s != "help")
809                    .collect()
810            })
811            .unwrap_or_default();
812        nodes.push(HelpNode {
813            path: path.clone(),
814            usage,
815            about,
816            sections,
817        });
818        for sub in subs {
819            let mut next = path.clone();
820            next.push(sub);
821            walk(root, bin, spec, next, nodes, seen)?;
822        }
823        Ok(())
824    }
825
826    walk(root, &bin_path, spec, Vec::new(), &mut nodes, &mut seen)?;
827    let name = nodes
828        .first()
829        .map(|n| prog_name(&n.usage))
830        .unwrap_or_else(|| "cli".to_string());
831    Ok(render_help_walk(&name, &nodes, spec.title.as_deref()))
832}

836fn gen_config(
837    repo: &Path,
838    config: Option<&Path>,
839    files: &[PathBuf],
840    only: &[String],
841    title: Option<&str>,
842    out: Option<&Path>,
843) -> Result<i32> {
844    let discovered = discover_config(repo, config)?;
845
846    // Legacy mode — explicit --file scans, generic page.
847    if !files.is_empty() {
848        let mut structs = Vec::new();
849        for f in files {
850            let src = std::fs::read_to_string(f)
851                .with_context(|| format!("cannot read {}", f.display()))?;
852            structs.extend(scan_config_structs(&src));
853        }
854        if !only.is_empty() {
855            structs.retain(|s| only.iter().any(|n| n == &s.name));
856            let found: Vec<&str> = structs.iter().map(|s| s.name.as_str()).collect();
857            for n in only {
858                if !found.contains(&n.as_str()) {
859                    bail!("--only {n}: no `pub struct {n}` found in the given files");
860                }
861            }
862        }
863        if structs.is_empty() {
864            bail!("no `pub struct` definitions found — pass --file <config.rs>");
865        }
866        let page = render_config_page(title.unwrap_or("Config reference"), &structs);
867        return emit(page, out).map(|_| 0);
868    }
869
870    // Curated mode — `[config]` in cypera.toml.
871    let Some((cfg_path, rc)) = &discovered else {
872        bail!(
873            "gen config: no --file given and no `[config]` section — \
874             pass --file <rs>… or add [config] to cypera.toml"
875        );
876    };
877    let Some(doc) = &rc.config else {
878        bail!(
879            "gen config: cypera.toml at {} has no [config] section",
880            cfg_path.display()
881        );
882    };
883    let root = repo
884        .canonicalize()
885        .with_context(|| format!("--repo {} does not exist", repo.display()))?;
886    let base = config_dir(&discovered);
887    let page = config_doc_page(&root, &base, doc, only, title)?;
888    // Same resolution rule as `[cli].out` — config-declared paths
889    // are repo-root-relative; `--out` stays cwd-relative.
890    let emit_to = match out {
891        Some(p) => Some(p.to_path_buf()),
892        None => doc.out.as_ref().map(|rel| resolve_path(&root, &base, rel)),
893    };
894    emit_to_path(page, emit_to.as_deref())?;
895    Ok(0)
896}

898/// Render the `[config]`-driven page — shared by `gen config` and
899/// `check` freshness.
900fn config_doc_page(
901    root: &Path,
902    base: &Path,
903    doc: &ConfigDocConfig,
904    only: &[String],
905    title_override: Option<&str>,
906) -> Result<String> {
907    // Scan every declared source file once; index structs by name.
908    let mut scanned: std::collections::HashMap<String, ConfigStruct> =
909        std::collections::HashMap::new();
910    for sref in &doc.structs {
911        let path = resolve_path(root, base, &sref.file);
912        let src = std::fs::read_to_string(&path)
913            .with_context(|| format!("cannot read {}", path.display()))?;
914        let mut found = false;
915        for s in scan_config_structs(&src) {
916            if s.name == sref.name {
917                let mut s = s;
918                if doc.all_optional.iter().any(|n| n == &sref.name) {
919                    for f in &mut s.fields {
920                        f.optional = true;
921                    }
922                }
923                scanned.insert(sref.name.clone(), s);
924                found = true;
925            }
926        }
927        if !found {
928            // Python parity: `fields()` returned [] for a missing
929            // struct — the section renders a header-only table.
930            // (tamias `ConfigFormat` is a trait, not a struct.)
931            eprintln!(
932                "cypera: warning — [config.structs] `{}`: no `pub struct` in {}",
933                sref.name,
934                path.display()
935            );
936            scanned.insert(
937                sref.name.clone(),
938                ConfigStruct {
939                    name: sref.name.clone(),
940                    doc: String::new(),
941                    fields: Vec::new(),
942                },
943            );
944        }
945    }
946
947    let sections: Vec<ConfigDocSection> = if doc.sections.is_empty() {
948        doc.structs
949            .iter()
950            .filter(|s| only.is_empty() || only.iter().any(|n| n == &s.name))
951            .map(|sref| ConfigDocSection {
952                level: 2,
953                title: Some(sref.title.clone()),
954                strukt: scanned.get(&sref.name).cloned(),
955                body: None,
956            })
957            .collect()
958    } else {
959        let mut sections = Vec::with_capacity(doc.sections.len());
960        for sec in &doc.sections {
961            // A section may combine a struct table with trailing
962            // prose (`struct` + `text`/`text_file`) — the tarsi
963            // sync-rule/ghreleases convention.
964            let body = match (&sec.text, &sec.text_file) {
965                (Some(t), _) => Some(t.clone()),
966                (None, Some(f)) => {
967                    let p = resolve_path(root, base, f);
968                    Some(
969                        std::fs::read_to_string(&p)
970                            .with_context(|| format!("cannot read text_file {}", p.display()))?,
971                    )
972                }
973                (None, None) => None,
974            };
975            if let Some(name) = &sec.struct_name {
976                if !only.is_empty() && !only.iter().any(|n| n == name) {
977                    continue;
978                }
979                sections.push(ConfigDocSection {
980                    level: sec.level,
981                    title: sec.title.clone(),
982                    strukt: scanned.get(name).cloned(),
983                    body,
984                });
985            } else {
986                sections.push(ConfigDocSection {
987                    level: sec.level,
988                    title: sec.title.clone(),
989                    strukt: None,
990                    body,
991                });
992            }
993        }
994        sections
995    };
996
997    let preamble = match (&doc.preamble, &doc.preamble_file) {
998        (Some(t), _) => Some(t.clone()),
999        (None, Some(f)) => {
1000            let p = resolve_path(root, base, f);
1001            Some(
1002                std::fs::read_to_string(&p)
1003                    .with_context(|| format!("cannot read preamble_file {}", p.display()))?,
1004            )
1005        }
1006        (None, None) => None,
1007    };
1008    let style = cypera_core::ConfigDocStyle {
1009        preamble,
1010        marker: doc.marker.clone(),
1011        table_head: doc.table_head.clone(),
1012        table_sep: doc.table_sep.clone(),
1013    };
1014    Ok(render_config_doc_page(
1015        title_override.unwrap_or(&doc.title),
1016        &sections,
1017        &doc.types,
1018        &doc.field_types,
1019        &style,
1020    ))
1021}

1025/// `<NAME>_ROOT` env override — NAME is the basename of the dir
1026/// holding `cypera.toml` (the repo root in the common case),
1027/// uppercased with non-alphanumerics folded to `_`. This is the
1028/// python `gen-api.py` `TAMIAS_ROOT`/`MYCO_ROOT`/`CUORA_ROOT`/
1029/// `PAVO_ROOT` convention generalized: under bupha the docs sha
1030/// runs detached and the env var points cargo at the real
1031/// checkout.
1032fn api_root_env_name(config_dir: &Path, repo: &Path) -> String {
1033    let base = if config_dir.as_os_str().is_empty() {
1034        repo
1035    } else {
1036        config_dir
1037    };
1038    base.file_name()
1039        .map(|n| n.to_string_lossy().into_owned())
1040        .unwrap_or_else(|| "repo".to_string())
1041        .chars()
1042        .map(|c| {
1043            if c.is_ascii_alphanumeric() {
1044                c.to_ascii_uppercase()
1045            } else {
1046                '_'
1047            }
1048        })
1049        .collect::<String>()
1050        + "_ROOT"
1051}

1053/// Recursive dir copy — the python `shutil.copytree` half of
1054/// `gen-api.py` (dirs recurse, everything else `fs::copy`).
1055fn copy_dir_all(src: &Path, dst: &Path) -> Result<()> {
1056    std::fs::create_dir_all(dst).with_context(|| format!("cannot create {}", dst.display()))?;
1057    for entry in std::fs::read_dir(src).with_context(|| format!("cannot read {}", src.display()))? {
1058        let entry = entry?;
1059        let s = entry.path();
1060        let d = dst.join(entry.file_name());
1061        // `is_dir` follows symlinks — python's copytree default.
1062        if s.is_dir() {
1063            copy_dir_all(&s, &d)?;
1064        } else {
1065            std::fs::copy(&s, &d).with_context(|| format!("cannot copy {}", s.display()))?;
1066        }
1067    }
1068    Ok(())
1069}

1071/// `cypera gen api` — the `docs/gen-api.py` port:
1072///
1073/// 1. `cargo doc --workspace --no-deps` in the repo root.
1074/// 2. `cargo metadata --no-deps --format-version 1` → the real
1075///    `target_directory` (respects `build.target-dir` and
1076///    `CARGO_TARGET_DIR`).
1077/// 3. Empty the api dir, copy `<target>/doc` into it.
1078/// 4. Write the styled `index.html` over the crate dirs whose
1079///    names start with the prefix (the `gen-api-index.py`
1080///    `startswith("<repo>")` filter).
1081fn gen_api(
1082    repo: &Path,
1083    config: Option<&Path>,
1084    out: Option<&Path>,
1085    target_dir_flag: Option<&Path>,
1086) -> Result<i32> {
1087    let discovered = discover_config(repo, config)?;
1088    let rc = discovered.as_ref().map(|(_, c)| c);
1089    let api_cfg = rc.map(|c| &c.api);
1090    let base = config_dir(&discovered);
1091
1092    // Repo root: the `<NAME>_ROOT` env override wins (python
1093    // parity — the scripts read `os.environ["<REPO>_ROOT"]`
1094    // before falling back to the checkout); else canonical
1095    // `--repo`. NAME comes from the cypera.toml dir when one was
1096    // found, else the repo dir itself.
1097    let repo_canon = repo
1098        .canonicalize()
1099        .with_context(|| format!("--repo {} does not exist", repo.display()))?;
1100    let env_base = if discovered.is_some() {
1101        base.clone()
1102    } else {
1103        repo_canon.clone()
1104    };
1105    let env_name = api_root_env_name(&env_base, &repo_canon);
1106    let mut root = repo_canon;
1107    if let Ok(v) = std::env::var(&env_name) {
1108        if !v.is_empty() {
1109            root = PathBuf::from(&v)
1110                .canonicalize()
1111                .with_context(|| format!("{env_name}={v}: not a directory"))?;
1112        }
1113    }
1114
1115    // CARGO_TARGET_DIR for the doc build — flag > `[api].target_dir`;
1116    // only exported when the env var is unset (the python
1117    // `os.environ.setdefault` semantics — an explicit env always
1118    // wins, and `cargo metadata` resolves it the same way).
1119    let target_dir = target_dir_flag.map(Path::to_path_buf).or_else(|| {
1120        api_cfg.and_then(|a| {
1121            a.target_dir.as_ref().map(|rel| {
1122                let p = Path::new(rel);
1123                if p.is_absolute() {
1124                    p.to_path_buf()
1125                } else {
1126                    root.join(p)
1127                }
1128            })
1129        })
1130    });
1131    let cargo_env = |cmd: &mut std::process::Command| {
1132        if std::env::var_os("CARGO_TARGET_DIR").is_none() {
1133            if let Some(td) = &target_dir {
1134                cmd.env("CARGO_TARGET_DIR", td);
1135            }
1136        }
1137    };
1138
1139    // 1. cargo doc --workspace --no-deps.
1140    let mut doc = std::process::Command::new("cargo");
1141    doc.args(["doc", "--workspace", "--no-deps"])
1142        .current_dir(&root);
1143    cargo_env(&mut doc);
1144    let out_doc = doc.output().with_context(|| "cannot spawn `cargo doc`")?;
1145    if !out_doc.status.success() {
1146        bail!(
1147            "cargo doc --workspace --no-deps failed in {}:\n{}",
1148            root.display(),
1149            String::from_utf8_lossy(&out_doc.stderr)
1150        );
1151    }
1152
1153    // 2. cargo metadata → target_directory.
1154    let mut meta = std::process::Command::new("cargo");
1155    meta.args(["metadata", "--no-deps", "--format-version", "1"])
1156        .current_dir(&root);
1157    cargo_env(&mut meta);
1158    let out_meta = meta
1159        .output()
1160        .with_context(|| "cannot spawn `cargo metadata`")?;
1161    if !out_meta.status.success() {
1162        bail!(
1163            "cargo metadata failed in {}:\n{}",
1164            root.display(),
1165            String::from_utf8_lossy(&out_meta.stderr)
1166        );
1167    }
1168    let meta_json: serde_json::Value =
1169        serde_json::from_slice(&out_meta.stdout).context("cannot parse `cargo metadata` output")?;
1170    let target_dir_str = meta_json["target_directory"]
1171        .as_str()
1172        .context("cargo metadata: no target_directory")?;
1173    let doc_dir = Path::new(target_dir_str).join("doc");
1174    if !doc_dir.is_dir() {
1175        bail!("cargo doc produced no {}", doc_dir.display());
1176    }
1177
1178    // 3. api dir — `--out` (cwd-relative) > `[api].out`
1179    // (repo-relative, default `docs/public/api`).
1180    let api_dir = match out {
1181        Some(p) => p.to_path_buf(),
1182        None => {
1183            let rel = api_cfg.and_then(|a| a.out.clone());
1184            match rel {
1185                Some(rel) => resolve_path(&root, &base, &rel),
1186                None => root.join("docs").join("public").join("api"),
1187            }
1188        }
1189    };
1190    std::fs::create_dir_all(&api_dir)
1191        .with_context(|| format!("cannot create {}", api_dir.display()))?;
1192    // Empty it first — the python `for c in api.iterdir(): rmtree/unlink`.
1193    for entry in
1194        std::fs::read_dir(&api_dir).with_context(|| format!("cannot read {}", api_dir.display()))?
1195    {
1196        let entry = entry?;
1197        let p = entry.path();
1198        if p.is_dir() {
1199            std::fs::remove_dir_all(&p)?;
1200        } else {
1201            std::fs::remove_file(&p)?;
1202        }
1203    }
1204    for entry in
1205        std::fs::read_dir(&doc_dir).with_context(|| format!("cannot read {}", doc_dir.display()))?
1206    {
1207        let entry = entry?;
1208        let s = entry.path();
1209        let d = api_dir.join(entry.file_name());
1210        if s.is_dir() {
1211            copy_dir_all(&s, &d)?;
1212        } else {
1213            std::fs::copy(&s, &d).with_context(|| format!("cannot copy {}", s.display()))?;
1214        }
1215    }
1216
1217    // 4. index — crate dirs (name starts with the prefix),
1218    // sorted, `_`→`-` labels; title `"{prefix} — Crate API"`.
1219    let prefix = api_cfg.and_then(|a| a.prefix.clone()).unwrap_or_else(|| {
1220        root.file_name()
1221            .map(|n| n.to_string_lossy().into_owned())
1222            .unwrap_or_else(|| "repo".to_string())
1223    });
1224    let title = api_cfg
1225        .and_then(|a| a.title.clone())
1226        .unwrap_or_else(|| format!("{prefix} — Crate API"));
1227    let mut crates: Vec<String> = std::fs::read_dir(&api_dir)
1228        .with_context(|| format!("cannot read {}", api_dir.display()))?
1229        .filter_map(|e| e.ok())
1230        .filter(|e| {
1231            let p = e.path();
1232            p.is_dir()
1233                && e.file_name()
1234                    .to_str()
1235                    .is_some_and(|n| n.starts_with(&prefix))
1236        })
1237        .map(|e| e.file_name().to_string_lossy().into_owned())
1238        .collect();
1239    crates.sort();
1240    let page = render_api_index(&title, &crates);
1241    let index_path = api_dir.join("index.html");
1242    std::fs::write(&index_path, page)
1243        .with_context(|| format!("cannot write {}", index_path.display()))?;
1244    println!("rustdoc api written to {}", api_dir.display());
1245    println!("api index: {} crates", crates.len());
1246    Ok(0)
1247}

1256fn emit_to_path(page: String, out: Option<&Path>) -> Result<()> {
1257    match out {
1258        Some(path) => {
1259            if let Some(parent) = path.parent() {
1260                if !parent.as_os_str().is_empty() {
1261                    std::fs::create_dir_all(parent)
1262                        .with_context(|| format!("cannot create {}", parent.display()))?;
1263                }
1264            }
1265            std::fs::write(path, page)
1266                .with_context(|| format!("cannot write {}", path.display()))?;
1267            println!("written to {}", path.display());
1268        }
1269        None => print!("{page}"),
1270    }
1271    Ok(())
1272}
