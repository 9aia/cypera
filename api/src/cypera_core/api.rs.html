1//! `gen api` — the `docs/gen-api.py` + `docs/gen-api-index.py`
2//! port's pure half: the styled rustdoc index page. The IO half
3//! (`cargo doc`, `cargo metadata`, the `<target>/doc` →
4//! `docs/public/api` copy) lives in `cypera-cli`; everything here
5//! is strings in, page out.
6
7/// Render `public/api/index.html` — byte-for-byte the python
8/// `gen-api-index.py` output: one `<li>` per crate dir (the
9/// caller has already prefix-filtered and sorted the names),
10/// `_` → `-` in the link label.
11pub fn render_api_index(title: &str, crates: &[String]) -> String {
12    let links = crates
13        .iter()
14        .map(|c| {
15            format!(
16                "  <li><a href=\"{c}/index.html\">{}</a></li>",
17                c.replace('_', "-")
18            )
19        })
20        .collect::<Vec<_>>()
21        .join("\n");
22    format!(
23        "<!doctype html><meta charset=utf-8>\
24         <title>{title}</title>\
25         <style>body{{background:#1b1b1f;color:#ddd;\
26         font-family:ui-monospace,monospace;\
27         max-width:640px;margin:4em auto;\
28         padding:0 1em}}\
29         a{{color:#a8b1ff;text-decoration:none}}\
30         a:hover{{text-decoration:underline}}\
31         li{{line-height:1.9}}\
32         h1{{font-size:1.3em}}</style>\
33         <h1>{title}</h1>\n<ul>\n{links}\n</ul>\n"
34    )
35}

37#[cfg(test)]
38mod tests {
39    #![allow(clippy::unwrap_used)]
40    use super::*;
41
42    /// The exact bytes `gen-api-index.py` writes for a two-crate
43    /// site — guards the port against drift.
44    #[test]
45    fn index_matches_python_output() {
46        let page = render_api_index(
47            "tamias — Crate API",
48            &["tamias".to_string(), "tamias_core".to_string()],
49        );
50        let expected = concat!(
51            "<!doctype html><meta charset=utf-8>",
52            "<title>tamias — Crate API</title>",
53            "<style>body{background:#1b1b1f;color:#ddd;font-family:",
54            "ui-monospace,monospace;max-width:640px;margin:4em auto;",
55            "padding:0 1em}a{color:#a8b1ff;text-decoration:none}",
56            "a:hover{text-decoration:underline}li{line-height:1.9}",
57            "h1{font-size:1.3em}</style>",
58            "<h1>tamias — Crate API</h1>\n<ul>\n",
59            "  <li><a href=\"tamias/index.html\">tamias</a></li>\n",
60            "  <li><a href=\"tamias_core/index.html\">tamias-core</a></li>",
61            "\n</ul>\n",
62        );
63        assert_eq!(page, expected);
64    }
65
66    #[test]
67    fn empty_site_still_renders_list_shell() {
68        let page = render_api_index("x — Crate API", &[]);
69        assert!(page.contains("<h1>x — Crate API</h1>\n<ul>\n\n</ul>\n"));
70    }
